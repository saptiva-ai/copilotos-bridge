```python
def _extract_metric_heuristic(self, user_query: str, intent_hint: Optional[str]) -> Optional[str]:
    # NUEVO: Try Weaviate first
    metric = self._get_metric_from_ontology(user_query)
    if metric:
        return metric

    # Fallback to hardcoded (deprecated, para backward compatibility)
    if intent_hint:
        metric = self.METRIC_MAP.get(intent_hint.lower())
        if metric:
            logger.warning("Using deprecated hardcoded METRIC_MAP")
            return metric

    return None

def _get_metric_from_ontology(self, query: str) -> Optional[str]:
    """Query Ontology_Term_V2 in Weaviate for metric matching."""
    from services.weaviate_service import get_weaviate_service

    weaviate_svc = get_weaviate_service()
    if not weaviate_svc.client.is_connected():
        logger.warning("Weaviate not available, skipping ontology lookup")
        return None

    # Hybrid search en Ontology_Term_V2
    collection = weaviate_svc.client.collections.get("Ontology_Term_V2")
    response = collection.query.hybrid(
        query=query,
        limit=5,
        return_properties=["name", "synonyms", "linked_field"]
    )

    for obj in response.objects:
        term_name = obj.properties.get("name", "").lower()
        synonyms = obj.properties.get("synonyms", [])

        # Check if query contains term or synonym
        query_lower = query.lower()
        if term_name in query_lower:
            return obj.properties.get("linked_field")

        for syn in synonyms:
            if syn.lower() in query_lower:
                return obj.properties.get("linked_field")

    return None
```

```python
#!/usr/bin/env python3
"""
PoC QuerySpec Day 6 - Gate Bloqueante
======================================

Valida que QuerySpec Builder alcanza ≥90% precisión.

Criterios GO/NO-GO:
- ≥90% QuerySpec válidos (JSON bien formado)
- ≥90% alineados con ontología (no columnas inventadas)
- 0% SQL destructivo (only SELECT)

Uso:
    python scripts/poc/test_queryspec_day6.py

import json
import sys
from pathlib import Path
from typing import Dict, List
from dataclasses import dataclass

sys.path.insert(0, str(Path(__file__).parent.parent / "src"))
from bankadvisor.services.query_spec_parser import QuerySpecParser
from schemas.query_spec_validator import validate_query_spec, QuerySpecError

@dataclass
class TestResult:
    query_id: int
    query: str
    valid_json: bool
    aligned_ontology: bool
    sql_safe: bool
    error: str = ""

def load_dataset() -> List[Dict]:
    dataset_path = Path(__file__).parent / "queryspec_test_dataset.json"
    return json.loads(dataset_path.read_text())

def test_queryspec_builder():
    parser = QuerySpecParser()
    dataset = load_dataset()
    results = []

    for test_case in dataset:
        query = test_case["query"]
        expected = test_case.get("expected", {})

        try:
            # Parse query
            spec = await parser.parse(query)

            # Test 1: Valid JSON (well-formed QuerySpec)
            valid_json = spec is not None and spec.metric != ""

            # Test 2: Aligned with ontology (no invented columns)
            # Check if metric exists in Ontology_Term_V2
            aligned = check_metric_in_ontology(spec.metric)

            # Test 3: SQL safe (if intent is SQL_QUERY)
            sql_safe = True
            if spec.intent == "SQL_QUERY":
                try:
                    validate_query_spec(spec.to_dict())
                except QuerySpecError as e:
                    sql_safe = False
                    error = str(e)

            results.append(TestResult(
                query_id=test_case["id"],
                query=query,
                valid_json=valid_json,
                aligned_ontology=aligned,
                sql_safe=sql_safe
            ))

        except Exception as e:
            results.append(TestResult(
                query_id=test_case["id"],
                query=query,
                valid_json=False,
                aligned_ontology=False,
                sql_safe=False,
                error=str(e)
            ))

    # Calculate metrics
    total = len(results)
    valid_count = sum(1 for r in results if r.valid_json)
    aligned_count = sum(1 for r in results if r.aligned_ontology)
    safe_count = sum(1 for r in results if r.sql_safe)

    valid_pct = (valid_count / total) * 100
    aligned_pct = (aligned_count / total) * 100
    unsafe_count = total - safe_count
    unsafe_pct = (unsafe_count / total) * 100

    # Print results
    print("=" * 60)
    print("PoC QuerySpec Day 6 - Resultados")
    print("=" * 60)
    print(f"\nTotal queries: {total}")
    print(f"\n✅ QuerySpec válidos: {valid_count}/{total} ({valid_pct:.1f}%)")
    print(f"✅ Alineados con ontología: {aligned_count}/{total} ({aligned_pct:.1f}%)")
    print(f"✅ SQL seguro: {safe_count}/{total} ({100 - unsafe_pct:.1f}%)")
    print(f"❌ SQL destructivo: {unsafe_count}/{total} ({unsafe_pct:.1f}%)")

    print("\n" + "=" * 60)
    print("GATE DECISION")
    print("=" * 60)

    gate_passed = (
        valid_pct >= 90.0 and
        aligned_pct >= 90.0 and
        unsafe_pct == 0.0

        print("\nCriterios cumplidos:")

        print(f"  - Alineados: {aligned_pct:.1f}% >= 90%")

            print(f"  - SQL destructivo: {unsafe_pct:.1f}% > 0%")

if __name__ == "__main__":
```

```python
def validate_query_spec_full(spec: Dict) -> ValidationResult:
    """
    Validación en 3 capas (Arquitectura v1.2).

    Layer 1: Intent/Metric validation
    Layer 2: JSON Schema validation
    Layer 3: SQL safety validation
    """
    errors = []

    # Layer 1: Intent validation
    try:
        validate_intent(spec)
    except IntentValidationError as e:
        errors.append(f"Intent validation failed: {e}")

    # Layer 2: JSON Schema
    try:
        validate_schema(spec)
    except SchemaValidationError as e:
        errors.append(f"Schema validation failed: {e}")

    # Layer 3: SQL safety (existing code)
    try:
        validate_query_spec(spec)
    except QuerySpecError as e:
        errors.append(f"SQL validation failed: {e}")

    if errors:
        raise ValidationError(errors)

    return ValidationResult(valid=True, errors=[])
```
